cmake_minimum_required(VERSION 3.16)
project(ieskf_voxel_lio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LIO_BUILD_ROS "Build the ROS 2 adapter and executable" ON)
include(CTest)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Threads REQUIRED)
add_library(lio_core src/Preprocess.cc src/ImuProcessing.cc src/Ieskf.cc src/VoxelMap.cc src/LieUtils.cc)
target_include_directories(lio_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(lio_core PUBLIC Eigen3::Eigen Threads::Threads)
target_compile_features(lio_core PUBLIC cxx_std_17)
target_compile_options(lio_core PRIVATE -Wall -Wextra -Wpedantic)
if(LIO_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(diagnostic_msgs REQUIRED)
  find_package(visualization_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(livox_ros_driver2 REQUIRED)
  find_package(PCL REQUIRED COMPONENTS common)
  add_library(lio_ros src/RosWrapper.cc)
  target_link_libraries(lio_ros PUBLIC lio_core ${PCL_LIBRARIES})
  target_include_directories(lio_ros SYSTEM PUBLIC ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(lio_ros PUBLIC rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs visualization_msgs tf2_ros livox_ros_driver2)
  target_compile_options(lio_ros PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(lio_node src/Main.cc)
  target_link_libraries(lio_node PRIVATE lio_ros lio_core)
  target_compile_options(lio_node PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS lio_core lio_ros lio_node EXPORT lio_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})
  install(FILES README.md DESTINATION share/${PROJECT_NAME})
  ament_export_targets(lio_targets HAS_LIBRARY_TARGET)
  ament_export_dependencies(Eigen3 PCL Threads rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs visualization_msgs tf2_ros livox_ros_driver2)
endif()
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(lio_core_test tests/LieUtilsTest.cc tests/PreprocessTest.cc tests/ImuProcessingTest.cc tests/IeskfTest.cc tests/VoxelMapTest.cc)
  target_link_libraries(lio_core_test PRIVATE lio_core GTest::gtest_main)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test_results/${PROJECT_NAME}")
  add_test(NAME lio_core_test COMMAND lio_core_test "--gtest_output=xml:${CMAKE_CURRENT_BINARY_DIR}/test_results/${PROJECT_NAME}/lio_core_test.xml")
  if(LIO_BUILD_ROS)
    add_executable(lio_ros_test tests/RosWrapperTest.cc)
    target_link_libraries(lio_ros_test PRIVATE lio_ros GTest::gtest_main)
    add_test(NAME lio_ros_test COMMAND lio_ros_test "--gtest_output=xml:${CMAKE_CURRENT_BINARY_DIR}/test_results/${PROJECT_NAME}/lio_ros_test.xml")
  endif()
endif()
if(LIO_BUILD_ROS)
  ament_package()
endif()
